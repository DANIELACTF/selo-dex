 * A tela (pagina.html) faz todas as contas no navegador com o motor de
 * motor_js.html. Este arquivo só serve a página e guarda as simulações

var ABA_SIM = 'Simulações';
var ABA_RES = 'Resultado';

/** A planilha do próprio script; se o script não estiver preso a uma, cria e lembra de uma. */
function planilha_(){
  var ativa = null;
  try{ ativa = SpreadsheetApp.getActiveSpreadsheet(); }catch(e){}
  if(ativa) return ativa;
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('planilha_id');
  if(id){
    try{ return SpreadsheetApp.openById(id); }catch(e){}
  }
  var nova = SpreadsheetApp.create('Simulador IBS/CBS — dados');
  props.setProperty('planilha_id', nova.getId());
  return nova;
}

function abaSimulacoes_(){
  var ss = planilha_();
  var aba = ss.getSheetByName(ABA_SIM);
  if(!aba){
    aba = ss.insertSheet(ABA_SIM);
    aba.getRange(1, 1, 1, CAB_SIM.length).setValues([CAB_SIM]).setFontWeight('bold');
    aba.getRange(1, COL_DADOS).setValue('Dados (não editar)').setFontWeight('bold');
    aba.setFrozenRows(1);
    aba.hideColumns(1);
    aba.getRange(1, 1, aba.getMaxRows(), CAB_SIM.length).setNumberFormat('@');
  }
  return aba;
}

function textoDe_(v){ return v === null || v === undefined ? '' : String(v).trim(); }

/** Texto digitado pelo usuário, sem sinais que o Sheets leria como fórmula. */
function semFormula_(v){ return textoDe_(v).replace(/^[=+\-@]+/, '').trim(); }

function acharLinha_(aba, id){
  var n = aba.getLastRow();
  if(n < 2) return 0;
  var col = aba.getRange(2, 1, n - 1, 1).getValues();
  for(var i = 0; i < col.length; i++){ if(textoDe_(col[i][0]) === id) return i + 2; }
  return 0;
}

function comTrava_(fn){
  var trava = LockService.getScriptLock();
  try{ trava.waitLock(20000); }
  catch(e){ throw new Error('A planilha está ocupada. Tente de novo em alguns segundos.'); }
  try{ return fn(); }
  finally{ try{ trava.releaseLock(); }catch(e){} }
}

function listar_(){
  var aba = abaSimulacoes_();
  var n = aba.getLastRow();
  if(n < 2) return [];
  return aba.getRange(2, 1, n - 1, CAB_SIM.length).getValues()
    .filter(function(l){ return textoDe_(l[0]); })
    .map(function(l){
      return { id: textoDe_(l[0]), nome: textoDe_(l[1]), empresa: textoDe_(l[2]), regime: textoDe_(l[3]),
               operacoes: Number(l[4]) || 0, salvaEm: textoDe_(l[5]) };
    })
    .sort(function(a, b){ return a.salvaEm < b.salvaEm ? 1 : -1; });
}

function agora_(){
  return Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm');
}

function salvarSimulacao(dados){
  dados = dados || {};
  var nome = semFormula_(dados.nome).slice(0, 120);
  if(!nome) throw new Error('Dê um nome à simulação.');
  var estado = dados.estado || {};
  var json = JSON.stringify(estado);
  var pedacos = [];
  for(var i = 0; i < json.length; i += PEDACO) pedacos.push(MARCA + json.slice(i, i + PEDACO));
  if(pedacos.length > MAX_PEDACOS) throw new Error('A simulação é grande demais para gravar. Reduza o número de operações.');

  return comTrava_(function(){
    var aba = abaSimulacoes_();
    var id = textoDe_(dados.id);
    var linha = id ? acharLinha_(aba, id) : 0;
    if(!linha){
      // Mesmo nome = mesma simulação: grava por cima em vez de duplicar.
      var lista = listar_().filter(function(s){ return s.nome.toLowerCase() === nome.toLowerCase(); });
      if(lista.length){ id = lista[0].id; linha = acharLinha_(aba, id); }
    }
    if(!id) id = Utilities.getUuid().replace(/-/g, '');
    if(!linha) linha = aba.getLastRow() + 1;

    var p = estado.parametros || {};
    var meta = [id, nome, semFormula_(p.empresa).slice(0, 120), p.regime === 'presumido' ? 'Lucro Presumido' : 'Lucro Real',
                (estado.operacoes || []).length, agora_()];
    aba.getRange(linha, 1, 1, meta.length).setValues([meta]);
    var larguraAntiga = Math.max(aba.getLastColumn() - COL_DADOS + 1, 1);
    aba.getRange(linha, COL_DADOS, 1, larguraAntiga).clearContent();
    aba.getRange(linha, COL_DADOS, 1, pedacos.length).setValues([pedacos]);
    return { id: id, simulacoes: listar_() };
  });
}

function abrirSimulacao(id){
  var aba = abaSimulacoes_();
  var linha = acharLinha_(aba, textoDe_(id));
  if(!linha) throw new Error('Simulação não encontrada.');
  var largura = Math.max(aba.getLastColumn() - COL_DADOS + 1, 1);
  var json = aba.getRange(linha, COL_DADOS, 1, largura).getValues()[0].map(function(v){
    var t = v === null || v === undefined ? '' : String(v);   // sem trim: o espaço pode ser parte do JSON
    return t.charAt(0) === MARCA ? t.slice(1) : t;
  }).join('');
  var nome = textoDe_(aba.getRange(linha, 2).getValue());
  try{
    return { id: textoDe_(id), nome: nome, estado: JSON.parse(json) };
  }catch(e){
    throw new Error('Os dados desta simulação foram alterados na planilha e não puderam ser lidos.');
  }
}

function excluirSimulacao(id){
  return comTrava_(function(){
    var aba = abaSimulacoes_();
    var linha = acharLinha_(aba, textoDe_(id));
    if(!linha) throw new Error('Simulação não encontrada.');
    aba.deleteRow(linha);
    return { simulacoes: listar_() };
  });
}

/**
 * Grava o resultado numa aba, para filtrar, imprimir ou baixar em Excel.
 * blocos: [{ titulo, cabecalho: [..], linhas: [[..]], formatos: ['texto'|'moeda'|'pct', ..] }]
 */
function gravarResultado(dados){
  dados = dados || {};
  var blocos = dados.blocos || [];
  return comTrava_(function(){
    var ss = planilha_();
    var aba = ss.getSheetByName(ABA_RES) || ss.insertSheet(ABA_RES);
    aba.clear();
    var linha = 1;
    aba.getRange(linha, 1).setValue(textoDe_(dados.titulo) || 'Simulação IBS/CBS').setFontWeight('bold').setFontSize(14);
    linha++;
    if(dados.subtitulo){ aba.getRange(linha, 1).setValue(textoDe_(dados.subtitulo)).setFontColor('#595959'); linha++; }
    linha++;

    var FMT = { moeda: '"R$" #,##0.00;[Red]-"R$" #,##0.00', pct: '0.00%', texto: '@' };
    blocos.forEach(function(b){
      var cab = b.cabecalho || [];
      var linhas = b.linhas || [];
      if(!cab.length) return;
      aba.getRange(linha, 1).setValue(textoDe_(b.titulo)).setFontWeight('bold');
      linha++;
      aba.getRange(linha, 1, 1, cab.length).setValues([cab]).setFontWeight('bold').setBackground('#e7eafb');
      linha++;
      if(linhas.length){
        (b.formatos || []).forEach(function(f, j){
          if(FMT[f]) aba.getRange(linha, j + 1, linhas.length, 1).setNumberFormat(FMT[f]);
        });
        aba.getRange(linha, 1, linhas.length, cab.length).setValues(linhas.map(function(l){
          return cab.map(function(_, j){ return l[j] === undefined || l[j] === null ? '' : l[j]; });
        }));
        linha += linhas.length;
      }
      linha++;
    });
    aba.autoResizeColumns(1, 15);
    return { url: ss.getUrl() + '#gid=' + aba.getSheetId() };
  });
}

function linkDaPlanilha(){
  return { url: planilha_().getUrl() };
}

      if(window.google && window.google.script && window.google.script.run) return ok();
      if(++tentativas > 50) return falha(new Error('Sem ligação com a planilha. As simulações ficam só neste navegador.'));
      setTimeout(ver, 100);
    })();
  });
}
async function chamar(nome, args){
  await prontoGoogle();
  return new Promise(function(ok, falha){
    google.script.run
      .withSuccessHandler(ok)
      .withFailureHandler(function(e){ falha(new Error((e && e.message) || 'Não foi possível concluir a operação.')); })
      [nome].apply(null, args || []);
  });
}

function numerosInvalidos(o){
  var ruins = [];
  if(isNaN(o.valor)) ruins.push('valor');
  [['icms', 'ICMS'], ['iss', 'ISS'], ['ipi', 'IPI'], ['is', 'Imposto Seletivo']].forEach(function(k){ if(isNaN(o[k[0]])) ruins.push(k[1]); });
  if(o.tratamento === 'personalizado' && isNaN(o.reducao)) ruins.push('redução');
  if(o.credPct !== null && isNaN(o.credPct)) ruins.push('crédito da compra');
  return ruins;
}

async function acaoSimulacoes(ev){
  var b = ev.target.closest('button');
  if(!b) return;
  try{
    if(b.dataset.abrir){
      if(E.operacoes.length && !(await confirmar('Abrir simulação', 'O que está na tela agora será substituído pela simulação salva.', 'Abrir'))) return;
      var d = await chamar('abrirSimulacao', [b.dataset.abrir]);
      aplicarEstado(Object.assign({}, d.estado, { simId: d.id, simNome: d.nome }));
      rascunho(); renderTudo();
      toast('Simulação "' + d.nome + '" aberta.');
    }
    if(b.dataset.apagar){
      var s = E.simulacoes.filter(function(x){ return x.id === b.dataset.apagar; })[0];
      if(!s || !(await confirmar('Excluir simulação', 'Excluir "' + s.nome + '" da planilha? Não dá para desfazer.', 'Excluir'))) return;
      var r = await chamar('excluirSimulacao', [s.id]);
      E.simulacoes = r.simulacoes;
      if(E.simId === s.id){ E.simId = ''; }
      rascunho(); renderSimulacoes();
      toast('Simulação excluída.');
    }
  }catch(e){ toast(e.message, true); }
}

function desenharGrafico(res){
  var max = res.reduce(function(m, r){ return Math.max(m, r.carga, 0); }, 0) || 1;
  var ref = Math.max(res[0].carga, 0) / max * 100;
  $('#grafico').innerHTML = '<div class="grafico">' + res.map(function(r){
    var partes = [
      ['atuais', Math.max(r.atuais, 0)],
      ['is', Math.max(r.is, 0)],
      ['ibscbs', r.efetivo ? Math.max(r.cbs + r.ibs, 0) : 0]
    ];
    var somaPos = partes.reduce(function(s, x){ return s + x[1]; }, 0);
    var fator = somaPos ? Math.max(r.carga, 0) / somaPos : 0;
    var titulo = r.ano + ': ' + M.moeda(r.carga) + ' (tributos atuais ' + M.moeda(r.atuais) + ', Seletivo ' + M.moeda(r.is) +
      ', IBS+CBS ' + M.moeda(r.cbs + r.ibs) + (r.efetivo || r.ano === 'Atual' ? '' : ', só destacados') + ')';
    return '<div class="g-linha" title="' + esc(titulo) + '">' +
      '<div class="g-nome">' + esc(r.ano) + (r.ano === '2026' ? '<span>teste</span>' : r.ano === 'Atual' ? '<span>hoje</span>' : '') + '</div>' +
      '<div class="g-faixa">' + partes.map(function(x){
        var w = x[1] * fator / max * 100;
        return w > 0 ? '<div class="g-seg ' + x[0] + '" style="width:' + w.toFixed(3) + '%"></div>' : '';
      }).join('') + '<div class="g-ref" style="left:' + ref.toFixed(3) + '%"></div></div>' +
      '<div class="g-valor">' + esc(M.moeda(r.carga)) + '<span>' + esc(M.pct(r.cargaPct)) +
        (r.ano === 'Atual' ? '' : ' · ' + esc(sinalPct(r.variacaoPct))) + '</span></div></div>';
  }).join('') + '</div>';
}

  /* ------------------------------------------------------------ cálculo */

  function aliquotasPisCofins(op, t, p){
    if(t.exportacao || op.pisCofins === 'monofasico' || op.pisCofins === 'isento') return { pis: 0, cofins: 0 };
    // Os créditos do não cumulativo usam sempre as alíquotas do regime.
    if(op.pisCofins === 'cumulativo' && op.tipo === 'venda') return { pis: 0.0065, cofins: 0.03 };
    var r = REGIMES[p.regime] || REGIMES.real;
    return { pis: r.pis, cofins: r.cofins };
  }

  function aliquotasIbsCbs(t, c){
    if(t.exportacao) return { cbs: 0, ibs: 0 };
    if(t.especial === 'financeiro'){
      // Alíquota total do regime financeiro, repartida na proporção CBS/IBS do ano.
      var soma = c.cbs + c.ibs, parte = soma ? c.cbs / soma : 0;
      return { cbs: c.financeiro * parte, ibs: c.financeiro * (1 - parte) };
    }
    return { cbs: c.cbs * (1 - t.reducao), ibs: c.ibs * (1 - t.reducao) };
  }

  /* Crédito de compra do Simples ou de produtor rural: percentual do modelo
     definitivo, escalonado pela alíquota de IBS+CBS do ano. */
  function creditoPercentual(op, p, c, padrao){
    var pct = op.credPct !== null && op.credPct !== undefined && op.credPct !== '' ? Number(op.credPct) : padrao;
    var ref = p.cbsRef + p.ibsRef, soma = c.cbs + c.ibs;
    if(!pct || !ref || !soma) return { cbs: 0, ibs: 0 };
    var total = op.valor * pct * Math.min(soma / ref, 1);
    return { cbs: total * c.cbs / soma, ibs: total * c.ibs / soma };
  }

  function calcularCenario(ops, p, c){
    var r = { ano: String(c.ano), faturamento: 0, compras: 0, icms: 0, iss: 0, ipi: 0, pis: 0, cofins: 0, is: 0,
              cbsDebito: 0, cbsCredito: 0, ibsDebito: 0, ibsCredito: 0, porTratamento: {},
              // para a análise de preço e margem
              tribVendas: 0, ipiVendas: 0, credAtuais: 0, ipiCompras: 0, ibsCbsCompras: 0,
              vendasServico: 0 };
    var naoCumulativo = p.regime !== 'presumido';

    ops.forEach(function(op){
      var t = tratamentoDe(op), valor = Number(op.valor) || 0;
      if(!(valor > 0)) return;
      var exp = t.exportacao;
      var icms = exp ? 0 : valor * op.icms * c.fator;
      var iss = exp ? 0 : valor * op.iss * c.fator;
      var ipi = c.ipi && !exp ? valor * op.ipi : 0;
      var isv = c.is && !exp ? valor * op.is : 0;
      var apc = aliquotasPisCofins(op, t, p);
      var basePC = valor - (p.excluiIcmsPC ? icms : 0);
      var pis = c.pisCofins ? basePC * apc.pis : 0;
      var cofins = c.pisCofins ? basePC * apc.cofins : 0;
      var base = valor - (p.excluiTribBase ? icms + iss + pis + cofins : 0) + isv;
      var aliq = aliquotasIbsCbs(t, c);
      var cbs = base * aliq.cbs, ibs = base * aliq.ibs;

      var pt = r.porTratamento[t.id] || (r.porTratamento[t.id] = {
        id: t.id, nome: t.nome, rotulo: rotuloReducao(t), vendas: 0, debito: 0, compras: 0, credito: 0 });

      if(op.tipo === 'venda'){
        r.faturamento += valor;
        if(t.servico || op.iss > 0) r.vendasServico += valor;
        r.icms += icms; r.iss += iss; r.ipi += ipi; r.is += isv; r.pis += pis; r.cofins += cofins;
        r.tribVendas += icms + iss + pis + cofins;
        r.ipiVendas += ipi;
        r.cbsDebito += cbs; r.ibsDebito += ibs;
        pt.vendas += valor; pt.debito += cbs + ibs;
        return;
      }

      r.compras += valor;
      r.ipiCompras += ipi;
      if(op.credIcmsIpi){ r.icms -= icms; r.ipi -= ipi; r.credAtuais += icms + ipi; }
      if(op.credPisCofins && naoCumulativo){ r.pis -= pis; r.cofins -= cofins; r.credAtuais += pis + cofins; }
      // Fornecedor do regime regular cobra IBS/CBS por fora; do Simples e do produtor rural, o preço já embute.
      if(op.fornecedor === 'regular') r.ibsCbsCompras += cbs + ibs;
      var cred = { cbs: 0, ibs: 0 };
      if(op.credIbsCbs && op.fornecedor !== 'sem'){
        if(op.fornecedor === 'simples') cred = creditoPercentual(op, p, c, p.credSimples);
        else if(op.fornecedor === 'rural') cred = creditoPercentual(op, p, c, p.credRural);
        else if(t.credito) cred = { cbs: cbs, ibs: ibs };
      }
      r.cbsCredito += cred.cbs; r.ibsCredito += cred.ibs;
      pt.compras += valor; pt.credito += cred.cbs + cred.ibs;
    });

    // ISS fixo da sociedade de profissionais: devido por profissional, não pelo faturamento.
    var issFixo = (Number(p.issFixo) || 0) * c.fator;
    r.iss += issFixo; r.tribVendas += issFixo;

    r.cbs = r.cbsDebito - r.cbsCredito;
    r.ibs = r.ibsDebito - r.ibsCredito;
    r.efetivo = !!c.efetivo;
    r.atuais = r.icms + r.iss + r.ipi + r.pis + r.cofins;
    r.carga = r.atuais + r.is + (c.efetivo ? r.cbs + r.ibs : 0);
    r.cargaPct = r.faturamento ? r.carga / r.faturamento : 0;
    return r;
  }

  /** Cenário atual seguido de cada ano do cronograma, com a variação sobre o atual. */
  function simular(ops, p, cronograma){
    var res = [cenarioAtual()].concat(cronograma).map(function(c){ return calcularCenario(ops, p, c); });
    var base = res[0].carga;
    res.forEach(function(r){
      r.variacao = r.carga - base;
      r.variacaoPct = base ? r.variacao / Math.abs(base) : 0;
    });
    precoEMargem(res, p);
    return res;
  }

  /**
   * Preço e margem: o que sobra para a empresa e quanto o preço precisa mudar.
   *
   * Com o preço de hoje mantido, o resultado antes de IRPJ/CSLL de cada ano é
   *   vendas − tributos "por dentro" das vendas (ICMS, ISS, PIS, COFINS)
   *   − compras − IPI e IBS/CBS cobrados por fora nas compras + créditos
   *   − folha − outros custos sem crédito.
   * IPI, Imposto Seletivo e IBS/CBS das vendas são cobrados do cliente e
   * repassados ao governo, então não mexem no resultado; mexem no preço que
   * o cliente paga. O "reajuste" é o quanto o valor das vendas precisa mudar
   * para o resultado ficar igual ao de hoje (as compras ficam como estão).
   */
  function precoEMargem(res, p){
    var folha = (Number(p.folha) || 0) + (Number(p.outrosCustos) || 0);
    var b2b = Math.min(Math.max(Number(p.b2b) || 0, 0), 1);
    res.forEach(function(r){
      var ibsLiquido = r.efetivo ? (r.cbsCredito + r.ibsCredito) - r.ibsCbsCompras : 0;
      r.custoFixo = r.compras + r.ipiCompras + folha - r.credAtuais - ibsLiquido;   // não muda com o preço
      r.receitaLiquida = r.faturamento - r.tribVendas;                                // muda com o preço
      r.resultado = r.receitaLiquida - r.custoFixo;
      r.margem = r.faturamento ? r.resultado / r.faturamento : 0;
      r.precoCliente = r.faturamento + r.ipiVendas + r.is + (r.efetivo ? r.cbsDebito + r.ibsDebito : 0);
      r.precoClienteCredito = r.faturamento + r.ipiVendas + r.is;   // o IBS/CBS volta como crédito
    });
    var hoje = res[0];
    res.forEach(function(r){
      var fator = r.receitaLiquida > 0 ? (hoje.resultado + r.custoFixo) / r.receitaLiquida : null;
      r.reajuste = fator === null ? null : fator - 1;
      if(fator === null || !hoje.precoCliente){
        r.varPrecoSemCredito = r.varPrecoComCredito = r.varPrecoMedio = null;
        return;
      }
      // Hoje o cliente paga o preço cheio (ISS e o PIS/COFINS cumulativo não voltam como crédito).
      r.varPrecoSemCredito = fator * r.precoCliente / hoje.precoCliente - 1;
      r.varPrecoComCredito = fator * r.precoClienteCredito / hoje.precoCliente - 1;
      r.varPrecoMedio = b2b * r.varPrecoComCredito + (1 - b2b) * r.varPrecoSemCredito;
    });
    return res;
  }

  /* ------------------------------------------------------------ formatação */

  function moeda(v){
    var n = Number(v) || 0;
    return (n < 0 ? '−' : '') + 'R$ ' + Math.abs(n).toFixed(2)
      .replace('.', ',').replace(/\B(?=(\d{3})+(?!\d))/g, '.');
  }
  function pct(v, casas){
    var n = (Number(v) || 0) * 100;
    return (n < 0 ? '−' : '') + Math.abs(n).toFixed(casas === undefined ? 2 : casas).replace('.', ',') + '%';
  }
  /** Lê "1.234,56", "1234.56", "18" ou "18%". */
  function numero(texto){
    if(typeof texto === 'number') return texto;
    var s = String(texto == null ? '' : texto).trim().replace(/\s|R\$|%/g, '');
    if(!s) return 0;
    if(s.indexOf(',') >= 0) s = s.replace(/\./g, '').replace(',', '.');
    var n = Number(s);
    return isNaN(n) ? NaN : n;
  }

  return {
    REGIMES: REGIMES, PIS_COFINS: PIS_COFINS, FORNECEDORES: FORNECEDORES, PERFIS: PERFIS,

  };
}
